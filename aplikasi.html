Koneksi Google Spreadsheet

Hubungkan web laporan keuangan dengan Google Sheet Anda secara real-time.

Tempelkan URL Web App dari penyetelan Apps Script di bawah ini.

Panduan Hubungkan ke Google Sheet:

  1. Buka Google Sheet baru. Ganti nama Sheet1 menjadi: Penjualan
  2. Klik menu Ekstensi (Extensions) > Apps Script.
  3. Hapus semua kode lama, lalu salin kode di bawah ini:
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Penjualan");
  if(!sheet) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  
  var headers = data[0];
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = row[j];
    }
    result.push(obj);
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Penjualan");
  if (!sheet) {
    sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("Penjualan");
    sheet.appendRow(["id", "invoiceNo", "date", "customer", "vehicle", "itemsJson", "totalRevenue", "totalHPP", "totalProfit", "paymentMethod", "notes"]);
  }
  
  var payload = JSON.parse(e.postData.contents);
  if (payload.action === "saveAll") {
    sheet.clear();
    sheet.appendRow(["id", "invoiceNo", "date", "customer", "vehicle", "itemsJson", "totalRevenue", "totalHPP", "totalProfit", "paymentMethod", "notes"]);
    
    payload.data.forEach(function(tx) {
      sheet.appendRow([
        tx.id,
        tx.invoiceNo,
        tx.date,
        tx.customer,
        tx.vehicle,
        JSON.stringify(tx.items),
        tx.totalRevenue,
        tx.totalHPP,
        tx.totalProfit,
        tx.paymentMethod,
        tx.notes || ""
      ]);
    });
    return ContentService.createTextOutput(JSON.stringify({status: "success", count: payload.data.length})).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Invalid action"})).setMimeType(ContentService.MimeType.JSON);
}

  2. Ubah Who has access menjadi: Anyone (Siapa saja).
  3. Klik Deploy, berikan otorisasi, lalu tempel Web App URL ke kolom di atas!
